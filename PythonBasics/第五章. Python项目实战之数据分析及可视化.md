## **某连锁店销售订单统计分析**

- 需求：

    - 需求1：统计每天销售额的变化。（折线图）

    - 需求2：统计对比不同城市的累计销售数量。（柱状图）

    - 需求3：统计不同产品类型对应的订单比例。（饼状图）

    - 需求4：同步不同支付方式对应的订单比例。（饼状图）



- 代码如下:

```Python
"""
某连锁商城销售数据统计分析
从CSV文件读取销售数据，生成统计图表
"""

import pandas as pd
import matplotlib.pyplot as plt
from pylab import mpl


def load_data(file_path: str) -> pd.DataFrame:
    """加载销售数据"""
    return pd.read_csv(
        file_path,
        encoding='utf-8',
        usecols=['订单日期', '产品类别', '产品名称', '销售数量', '单价', '客户所在城市', '支付方式']
    )


def process_category_data(data: pd.DataFrame):
    """处理产品类别数据，统计不同产品类型的订单数量"""
    category_count = data['产品类别'].value_counts()
    return category_count.index, category_count.values


def process_daily_sales_data(data: pd.DataFrame):
    """处理每天销售额数据（销售数量 * 单价）"""
    # 计算每条订单的销售额
    data['销售额'] = data['销售数量'] * data['单价']
    
    # 按日期分组统计每天销售额
    daily_sales = data.groupby('订单日期')['销售额'].sum().sort_index()
    return daily_sales.index, daily_sales.values


def process_city_data(data: pd.DataFrame):
    """处理城市数据，统计各个城市的销售数量"""
    city_sales = data.groupby('客户所在城市')['销售数量'].sum().sort_values(ascending=False)
    return city_sales.index, city_sales.values


def process_payment_data(data: pd.DataFrame):
    """处理支付方式数据，统计各支付方式对应的订单数量"""
    payment_count = data['支付方式'].value_counts()
    return payment_count.index, payment_count.values


def setup_chinese_font():
    """设置中文字体"""
    mpl.rcParams["font.sans-serif"] = ["SimHei"]
    mpl.rcParams["axes.unicode_minus"] = False


def create_plots(data: pd.DataFrame, output_path: str = None):
    """创建统计图表"""
    # 创建画布
    fig, axes = plt.subplots(2, 2, figsize=(16, 12), dpi=100)
    fig.suptitle('商城销售数据统计分析', fontsize=20)

    axes1, axes2, axes3, axes4 = axes[0, 0], axes[0, 1], axes[1, 0], axes[1, 1]

    # 图1: 每天销售额变化 - 折线图
    x_date, y_sales = process_daily_sales_data(data)
    axes1.plot(range(len(x_date)), y_sales, color='orange', linewidth=1.5, linestyle='-', markersize=3)
    axes1.set_title('每天销售额变化', fontsize=14)
    axes1.set_xlabel('日期')
    axes1.set_ylabel('销售额（元）')
    
    # 设置x轴刻度，每隔一定天数显示一个日期
    step = max(1, len(x_date) // 20)
    axes1.set_xticks(range(0, len(x_date), step))
    axes1.set_xticklabels([x_date[i] for i in range(0, len(x_date), step)], rotation=45, ha='right')
    axes1.grid(True, linestyle='--', alpha=0.3)

    # 图2: 各个城市的销售数量统计 - 柱状图
    x_city, y_city_count = process_city_data(data)
    axes2.bar(x_city, y_city_count, color='orange', width=0.6)
    axes2.tick_params(axis='x', rotation=90)
    axes2.set_title('各个城市的销售数量统计', fontsize=14)
    axes2.set_xlabel('城市')
    axes2.set_ylabel('销售数量')
    axes2.grid(True, linestyle='--', alpha=0.3)

    # 图3: 不同产品类型对应的订单数量 - 饼状图
    x_category, y_category_count = process_category_data(data)
    colors1 = ['#ff9999', '#66b3ff', '#99ff99', '#ffcc99', '#ff99cc', '#c2c2f0']
    axes3.pie(y_category_count, labels=x_category, autopct='%1.1f%%',
              startangle=90, colors=colors1, radius=1.0)
    axes3.set_title('不同产品类型对应的订单数量', fontsize=14)
    axes3.legend(loc='lower center', bbox_to_anchor=(0.5, -0.1),
                 ncol=len(x_category), fontsize=10)

    # 图4: 各支付方式对应的订单数量 - 饼状图
    x_payment, y_payment_count = process_payment_data(data)
    colors = ['#ff9999', '#66b3ff', '#99ff99', '#ffcc99']
    axes4.pie(y_payment_count, labels=x_payment, autopct='%1.1f%%',
              startangle=90, colors=colors, radius=1.0)
    axes4.set_title('各支付方式对应的订单数量', fontsize=14)
    axes4.legend(loc='lower center', bbox_to_anchor=(0.5, -0.1),
                 ncol=len(x_payment), fontsize=10)

    # 调整布局
    plt.tight_layout()
    plt.subplots_adjust(wspace=0.25, hspace=0.45, top=0.85)

    # 保存图片
    if output_path:
        plt.savefig(output_path, dpi=100, bbox_inches='tight')
        print(f"图表已保存至: {output_path}")

    # 显示图表
    plt.show()


def main():
    """主函数"""
    # 设置中文字体
    setup_chinese_font()

    # 数据文件路径
    data_file = 'data/sales.csv'
    output_file = 'data/sales_analysis.png'

    # 加载数据
    print(f"正在加载数据: {data_file}")
    data = load_data(data_file)
    print(f"成功加载 {len(data)} 条销售数据")
    
    # 显示数据基本信息
    print("\n数据预览:")
    print(data.head())
    print("\n产品类别统计:")
    print(data['产品类别'].value_counts())
    print("\n支付方式统计:")
    print(data['支付方式'].value_counts())

    # 创建图表
    create_plots(data, output_path=output_file)


if __name__ == '__main__':
    main()
```
